### Was bereits funktioniert (NICHT ANFASSEN):
- Komplettes React-Dashboard mit 8 Seiten
- Express-Backend mit CRUD für Agents und Calls (Drizzle ORM + PostgreSQL)
- Drizzle DB-Schema für agents, calls, organizations
- 4-Schritt Agent-Erstellungs-Wizard im Frontend
- Billing UI mit 3 Plan-Tiers
- pnpm-Workspace, TypeScript, Zod-Validierung, API-Client generiert aus OpenAPI

### Was FEHLT und implementiert werden muss:
1. **Vapi Integration** — Kein echter Assistent wird erstellt wenn Agent gespeichert wird
2. **Telnyx Integration** — Keine Telefonnummer wird gekauft/zugewiesen
3. **Vapi Webhook Handler** — call-started/call-ended existiert aber führt keine echten Aktionen aus
4. **Google Calendar Integration** — Kein echter Slot-Check, keine Buchung
5. **SMS Bestätigung** — Keine SMS nach Buchung
6. **Stripe Checkout** — Gibt nur Demo-URL zurück, kein echtes Payment
7. **Auth (Clerk)** — Kein Login, alle API-Routen sind öffentlich

import Telnyx from "telnyx";

const telnyx = process.env.TELNYX_API_KEY
  ? new Telnyx(process.env.TELNYX_API_KEY)
  : null;

export async function purchaseGermanPhoneNumber(): Promise<string | null> {
  if (!telnyx || !process.env.TELNYX_CONNECTION_ID) {
    console.warn("Telnyx not configured — skipping phone number purchase");
    return null;
  }

  try {
    // Verfügbare DE-Nummern suchen
    const { data: available } = await (telnyx.availablePhoneNumbers as any).list({
      filter: {
        country_code: "DE",
        features: ["voice"],
        limit: 3,
      },
    });

    if (!available?.length) {
      console.error("No German phone numbers available");
      return null;
    }

    // Erste verfügbare Nummer kaufen
    const { data: order } = await (telnyx.numberOrders as any).create({
      phone_numbers: [{ phone_number: available[0].phone_number }],
      connection_id: process.env.TELNYX_CONNECTION_ID,
    });

    return order?.phone_numbers?.[0]?.phone_number ?? null;
  } catch (err) {
    console.error("Failed to purchase Telnyx number:", err);
    return null;
  }
}

export async function sendSMS(to: string, message: string): Promise<boolean> {
  if (!telnyx || !process.env.TELNYX_SENDER_NUMBER) {
    console.warn("Telnyx not configured — SMS not sent");
    return false;
  }

  try {
    await (telnyx.messages as any).create({
      from: process.env.TELNYX_SENDER_NUMBER,
      to,
      text: message,
    });
    return true;
  } catch (err) {
    console.error("SMS send failed:", err);
    return false;
  }

### Schritt 4: Vapi Webhook Handler erweitern

**Ersetze `artifacts/api-server/src/routes/webhooks.ts` vollständig:**

```typescript
import { Router, type Request, type Response } from "express";
import { db, callsTable, agentsTable, organizationsTable } from "@workspace/db";
import { eq, sql } from "drizzle-orm";
import { sendSMS } from "../lib/telnyx.js";

const router = Router();

// POST /api/webhooks/vapi
router.post("/vapi", async (req: Request, res: Response) => {
  const secret = req.headers["x-vapi-secret"];
  if (process.env.VAPI_WEBHOOK_SECRET && secret !== process.env.VAPI_WEBHOOK_SECRET) {
    return res.status(401).json({ error: "Unauthorized" });
  }

  const event = req.body;
  const messageType = event?.message?.type;

  try {
    switch (messageType) {
      case "call-started": {
        const { call } = event.message;

        const agentResult = await db
          .select({ id: agentsTable.id, orgId: agentsTable.id })
          .from(agentsTable)
          .where(eq(agentsTable.vapiAssistantId, call.assistantId))
          .limit(1);

        await db.insert(callsTable).values({
          vapiCallId: call.id,
          agentId: agentResult[0]?.id ?? null,
          callerNumber: call.customer?.number ?? null,
          direction: "inbound",
          status: "in_progress",
          startedAt: new Date(),
        }).onConflictDoNothing();

        // Agent-Zähler inkrementieren
        if (agentResult[0]?.id) {
          await db
            .update(agentsTable)
            .set({ totalCalls: sql`${agentsTable.totalCalls} + 1` })
            .where(eq(agentsTable.id, agentResult[0].id));
        }
        break;
      }

      case "call-ended": {
        const { call } = event.message;
        const duration = call.endedAt && call.startedAt
          ? Math.round((new Date(call.endedAt).getTime() - new Date(call.startedAt).getTime()) / 1000)
          : 0;

        const minutesUsed = Math.max(1, Math.ceil(duration / 60));

        await db
          .update(callsTable)
          .set({
            status: "completed",
            durationSeconds: duration,
            endedAt: call.endedAt ? new Date(call.endedAt) : null,
            transcript: call.transcript ?? null,
            summary: call.summary ?? null,
            resolved: true,
            resolutionType: "completed",
          })
          .where(eq(callsTable.vapiCallId, call.id));

        // Minuten zur Organisation addieren
        const callRecord = await db
          .select({ agentId: callsTable.agentId })
          .from(callsTable)
          .where(eq(callsTable.vapiCallId, call.id))
          .limit(1);

        if (callRecord[0]?.agentId) {
          await db
            .update(agentsTable)
            .set({
              totalMinutes: sql`${agentsTable.totalMinutes} + ${minutesUsed}`,
            })
            .where(eq(agentsTable.id, callRecord[0].agentId));
        }
        break;
      }

      case "function-call": {
        // Tool Calls vom Vapi-Agent — werden über /api/tools/* Routen abgehandelt
        // Vapi erwartet eine direkte Antwort im Response Body
        const { functionCall } = event.message;
        console.log("Tool call received:", functionCall?.name, functionCall?.parameters);
        // Die eigentliche Tool-Logik ist in /api/tools/* implementiert
        // Hier nur ACK zurückgeben
        return res.json({ result: "Tool endpoint handles this separately" });
      }

      default:
        // Unbekannte Event-Typen stillschweigend ignorieren
        break;
    }

    res.json({ received: true });
  } catch (err) {
    console.error("Webhook error:", err);
    res.status(500).json({ error: "Internal server error" });
  }
});

export default router;
```

function generatePlaceholderSlots(date: string): string[] {
  const slots: string[] = [];
  const baseDate = new Date(date + "T00:00:00+02:00");

  if (isNaN(baseDate.getTime())) return [];

  const hours = [9, 10, 11, 14, 15, 16];
  for (const hour of hours) {
    const slot = new Date(baseDate);
    slot.setHours(hour, 0, 0, 0);
    if (slot > new Date()) {
      slots.push(slot.toISOString());
    }
  }

  return slots.slice(0, 3);
}

function formatGermanDateTime(isoString: string): string {
  try {
    return new Intl.DateTimeFormat("de-DE", {
      weekday: "long",
      day: "numeric",
      month: "long",
      hour: "2-digit",
      minute: "2-digit",
      timeZone: "Europe/Berlin",
    }).format(new Date(isoString));
  } catch {
    return isoString;
  }
}

export async function getAvailableSlots(
  accessToken: string,
  refreshToken: string,
  date: string,
  durationMinutes = 60
): Promise<string[]> {
  oauth2Client.setCredentials({ access_token: accessToken, refresh_token: refreshToken });
  const calendar = google.calendar({ version: "v3", auth: oauth2Client });

  const startOfDay = new Date(`${date}T08:00:00+02:00`);
  const endOfDay = new Date(`${date}T19:00:00+02:00`);

  const { data } = await calendar.freebusy.query({
    requestBody: {
      timeMin: startOfDay.toISOString(),
      timeMax: endOfDay.toISOString(),
      items: [{ id: "primary" }],
    },
  });

  const busyPeriods = data.calendars?.primary?.busy ?? [];
  const slotMs = durationMinutes * 60000;
  const slots: string[] = [];
  let cursor = new Date(startOfDay);

  while (cursor.getTime() + slotMs <= endOfDay.getTime()) {
    const slotEnd = new Date(cursor.getTime() + slotMs);
    const isConflict = busyPeriods.some((busy) => {
      return cursor < new Date(busy.end!) && slotEnd > new Date(busy.start!);
    });
    if (!isConflict) slots.push(cursor.toISOString());
    cursor = new Date(cursor.getTime() + slotMs);
  }

  return slots.slice(0, 3);
}

export async function createCalendarEvent(

  oauth2Client.setCredentials({ access_token: accessToken, refresh_token: refreshToken });
  const calendar = google.calendar({ version: "v3", auth: oauth2Client });

  const start = new Date(params.datetime);
  const end = new Date(start.getTime() + params.durationMinutes * 60000);

  const { data } = await calendar.events.insert({
    calendarId: "primary",
    requestBody: {
      summary: `${params.summary} — ${params.customerName}`,
      description: `📞 Gebucht via VoiceFlow AI\nKunde: ${params.customerName}\nTel: ${params.customerPhone}`,
      start: { dateTime: start.toISOString(), timeZone: "Europe/Berlin" },
      end: { dateTime: end.toISOString(), timeZone: "Europe/Berlin" },
      status: "confirmed",
    },
  });

  return data;
}

  }

  let event;

    event = stripe.webhooks.constructEvent(
      req.body, // raw body — braucht express.raw() Middleware
      sig as string,
      process.env.STRIPE_WEBHOOK_SECRET
    );

  }

  switch (event.type) {
    case "customer.subscription.created":
    case "customer.subscription.updated": {

      const orgId = sub.metadata?.orgId;
